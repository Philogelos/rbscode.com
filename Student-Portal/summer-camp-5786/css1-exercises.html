.game-button {

    /* COLORS */
    color: white;
    background-color: #5b45f0;

    /* SIZE */
    width: 200px;
    height: 55px;

    /* BORDER */
    border: none;
    border-radius: 14px;

    /* TEXT */
    font-size: 18px;
    font-weight: bold;
    font-family: Arial, sans-serif;
}
